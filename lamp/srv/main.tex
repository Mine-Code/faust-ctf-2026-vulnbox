\newcommand{\sessionCreate}{
  \begingroup
  \newread{\tmpfd}
  \immediate\openin\tmpfd=|/srv/gensessionid.sh
  \immediate\readline\tmpfd to \sessionId
  \immediate\closein\tmpfd
  \StrGobbleRight{\sessionId}{1}[\sessionId]
  \httpCookieAdd{Session=\sessionId}
  \redisSetEx{session_active_\sessionId}{1}{\sessionTimeout}
  \xdef\sessionId##1{\sessionId}
  \endgroup
}

\newcommand{\authMiddleware}[1]{
  \ifdefined\cookieSession
    \begingroup
    \redisGet{session_active_\cookieSession{}}{\sessionActive}
    \ifdefined\sessionActive
      \xdef\sessionId##1{\cookieSession{}}
      \redisGet{session_user_\cookieSession{}}{\username}
    \else
      % Session expired, create a new session
      \sessionCreate{}
    \fi
    \endgroup
  \else
    \sessionCreate{}
  \fi
  #1
}
