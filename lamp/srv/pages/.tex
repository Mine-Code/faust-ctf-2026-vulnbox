\newcommand{\actionAdd}{
    \sqlescape{\username{}}{\usernameEscaped}
    \sql{
        UPDATE ships SET numComponents = numComponents + 1 WHERE username = "\usernameEscaped";
    }{\ignore}
    \sql{
        SELECT id, numComponents FROM ships WHERE username = "\usernameEscaped";
    }{\result}
    \foreach \row in \result {
        \sqlparserow{\row}{rowdata}
        \sqlescape{\postx{}}{\xcoord}
        \sqlescape{\posty{}}{\ycoord}
        \sqlescape{\posttyp{}}{\type}
        \ifnum\rowdata[1]<50
            \sql{INSERT INTO components SET x = "\xcoord", y = "\ycoord", type = "\type", state = IF("\type" = "source", 1, 0), ship = \rowdata[0]}{\ignore}
        \fi
    }
}

\newcommand{\actionConnect}{
    \sqlescape{\username{}}{\usernameEscaped}
    \sql{
        SELECT id FROM ships WHERE username = "\usernameEscaped";
    }{\result}
    \foreach \row in \result {
        \sqlparserow{\row}{rowdata}
        \edef\id####1{\rowdata[0]}
        \sqlescape{\postleft{}}{\idleft}
        \sqlescape{\postright{}}{\idright}
        \sql{INSERT INTO connections (src, dst) SELECT "\idleft", "\idright" FROM components AS c1 JOIN components AS c2 ON c2.id = "\idright" WHERE c1.id = "\idleft" AND c1.ship = \id{} AND c2.ship = \id{}}{\ignore}
    }
}

\newcommand{\actionToggle}{
    \sqlescape{\username{}}{\usernameEscaped}
    \sql{
        SELECT id FROM ships WHERE username = "\usernameEscaped";
    }{\result}
    \foreach \row in \result {
        \sqlparserow{\row}{rowdata}
        \sqlescape{\postid{}}{\id}
        \edef\id####1{\id}
        \sql{UPDATE components SET properties = IF(properties = "on", "off", "on") WHERE id = "\id{}" AND type = "button" AND ship = \rowdata[0]}{\ignore}
    }

    % Tick one updates the button state
    \actionTick{}
    % Tick two updates connected components
    \actionTick{}
}

\newcommand{\actionTick}{
    \sqlescape{\username{}}{\usernameEscaped}
    \sql{
        SELECT id FROM ships WHERE username = "\usernameEscaped";
    }{\result}
    \foreach \row in \result {
        \sqlparserow{\row}{rowdata}
        \sql{UPDATE components c LEFT JOIN (SELECT w.dst AS id, MAX(s.state) AS anyOn FROM connections w JOIN components s ON s.id = w.src WHERE s.ship = \rowdata[0] GROUP BY w.dst) agg ON agg.id = c.id SET c.state = CASE c.type WHEN "light" THEN COALESCE(agg.anyOn, 0) WHEN "button" THEN IF(c.properties = "on" AND COALESCE(agg.anyOn, 0) = 1, 1, 0) ELSE c.state END WHERE c.ship = \rowdata[0] AND c.type <> "source"}{\ignore}
    }
}
